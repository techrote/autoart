nt(W*.52,H*.50,Math.min(W,H)*.12,W*.52,H*.50,Math.max(W,H)*.72);
    vg.addColorStop(.45,'rgba(0,0,0,0)');
    vg.addColorStop(1,'rgba(0,0,0,.31)');
    ctx.fillStyle=vg; ctx.fillRect(0,0,W,H);
  }

  function frame(now) {
    const dt = Math.min(.08, (now-last)/1000);
    last = now;
    if (!paused) {
      accumulator += dt;
      let guard = 0;
      while (accumulator >= STEP && guard++ < 5) {
        simulate();
        accumulator -= STEP;
      }
    }
    render();
    requestAnimationFrame(frame);
  }

  function reseed() {
    seed = hash32(seed + 0x9E3779B9);
    rebuild();
  }

  function savePng() {
    render();
    const a=document.createElement('a');
    a.download=`autoart-001-excavation-field-${seed.toString(16).padStart(8,'0')}.png`;
    a.href=canvas.toDataURL('image/png');
    a.click();
    flash.style.opacity='1';
    setTimeout(()=>flash.style.opacity='0',30);
  }

  addEventListener('resize', resize, {passive:true});
  addEventListener('pointermove', e => {
    pointer.x=e.clientX; pointer.y=e.clientY; pointer.active=true;
    pointer.strength=Math.min(1.35,pointer.strength+.10);
  }, {passive:true});
  addEventListener('pointerdown', e => {
    pointer.x=e.clientX; pointer.y=e.clientY; pointer.strength=1.7;
  }, {passive:true});
  addEventListener('pointerleave', () => pointer.active=false, {passive:true});
  addEventListener('keydown', e => {
    if (e.code === 'Space') { e.preventDefault(); paused=!paused; }
    else if (e.key === 'r' || e.key === 'R') reseed();
    else if (e.key === 's' || e.key === 'S') savePng();
    else if (e.key === 'h' || e.key === 'H') {
      showUI=!showUI; hud.classList.toggle('off',!showUI); keys.classList.toggle('off',!showUI);
    }
  });

  resize();
  requestAnimationFrame(frame);
})();
