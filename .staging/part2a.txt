        tctx.fillRect(sx, sy, g * .48, g * .48);
      }

      const out = p.x < -70 || p.x > W + 70 || p.y < -70 || p.y > H + 70;
      if (out || p.age > p.life) particles[i] = spawnParticle(p.id, false);
    }
    tctx.restore();
    tick++;
  }

  function makeBranches() {
    const list = [];
    const cx = W * .665, cy = H * .49;
    const baseR = Math.min(W,H) * .125;
    const local = makeRng(hash32(seed ^ 0xB12A7C55));

    function grow(x, y, a, len, depth, parent) {
      if (depth <= 0 || len < 4) return;
      const wobble = (local() - .5) * .42;
      const a2 = a + wobble;
      const x2 = x + Math.cos(a2) * len;
      const y2 = y + Math.sin(a2) * len;
      const node = { x1:x, y1:y, x2, y2, depth, parent, salt:local() };
      list.push(node);
      if (depth > 1) {
        grow(x2, y2, a2 + (local()-.5)*.28, len * randLocal(local,.71,.87), depth-1, node);
        if (local() < .58) grow(x2, y2, a2 + (local()<.5?-1:1)*randLocal(local,.28,.63), len * randLocal(local,.46,.68), depth-2, node);
      }
    }

    const roots = 9;
    for (let i=0;i<roots;i++) {
      const a = (i/roots)*Math.PI*2 + randLocal(local,-.16,.16);
      const x = cx + Math.cos(a)*baseR;
      const y = cy + Math.sin(a)*baseR*.78;
      grow(x, y, a + randLocal(local,-.30,.30), Math.min(W,H)*randLocal(local,.055,.11), 7, null);
    }
    return list;
  }

  function randLocal(r, a, b) { return a + (b-a)*r(); }
