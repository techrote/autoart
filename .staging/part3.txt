th=p.width;
      ctx.setLineDash(p.dash);
      ctx.lineDashOffset = -t * 12 * p.speed;
      ctx.stroke();
    }
    ctx.setLineDash([]);
    ctx.restore();
  }

  function drawSparseScheduler(t) {
    const spacing = clamp(Math.min(W,H)/42, 11, 22);
    const phase = Math.floor(t / 1.85) & 3;
    ctx.save();
    ctx.globalCompositeOperation='screen';
    for(let gy=0, y=spacing*.5; y<H; gy++,y+=spacing) {
      for(let gx=0, x=spacing*.5; x<W; gx++,x+=spacing) {
        if (((gx + gy*3) & 3) !== phase) continue;
        const mask = hash32((gx*73856093) ^ (gy*19349663) ^ seed) & 255;
        if(mask < 188) continue;
        const a = .018 + (mask/255)*.034;
        ctx.fillStyle=`rgba(118,211,255,${a})`;
        ctx.fillRect(x,y,1,1);
      }
    }
    ctx.restore();
  }

  function drawFaultBands(t) {
    ctx.save();
    const bands = 5;
    const bandH = Math.max(5, H * .018);
    for(let b=0;b<bands;b++) {
      const h = hash32(seed ^ (b*0x9E3779B9));
      const baseY = ((h & 0xffff)/0xffff) * (H-bandH);
      const y = clamp(baseY + Math.sin(t*.21 + b*1.7) * H*.025, 0, H-bandH);
      const drift = Math.sin(t*.37 + b*2.3) * (5 + (h>>>20 & 15));
      const alpha = .04 + ((h>>>8 & 255)/255)*.045;
      ctx.globalAlpha=alpha;
      ctx.globalCompositeOperation='screen';
      ctx.drawImage(trails, 0, y*DPR, trails.width, bandH*DPR, drift, y, W, bandH);
      ctx.fillStyle = b&1 ? 'rgba(255,60,195,.045)' : 'rgba(68,224,255,.045)';
      ctx.fillRect(0,y,W,1);
    }
    ctx.restore();
  }

  function drawBranches(t) {
    ctx.save();
    ctx.globalCompositeOperation='screen';
    const pulse = .72 + .28*Math.sin(t*.31);
    for (const b of branches) {
      const depthN = b.depth/7;
      const a = (.028 + .11*depthN) * pulse * (.65 + b.salt*.7);
      const c = b.salt < .52 ? PALETTE[0] : PALETTE[4];
      ctx.strokeStyle=`rgba(${c[0]},${c[1]},${c[2]},${a})`;
      ctx.lineWidth=.35 + depthN*.55;
      ctx.beginPath();
      ctx.moveTo(b.x1,b.y1); ctx.lineTo(b.x2,b.y2); ctx.stroke();
    }
    ctx.restore();
  }

  function drawGlyphBiopsy(t) {
    const data = glyphCells;
    const cell = clamp(Math.min(W,H)/48, 8, 15);
    const ox = W - data.cols*cell - Math.max(14,W*.025);
    const oy = H*.13;
    ctx.save();
    ctx.globalCompositeOperation='screen';
    ctx.font = `${cell*1.05}px ui-monospace, SFMono-Regular, Menlo, Consolas, monospace`;
    ctx.textBaseline='top';
    for(const c of data.cells) {
      const pulse=.5+.5*Math.sin(t*.24 + c.phase);
      const a=.025 + c.v*.09*pulse;
      const col = c.v>.68 ? PALETTE[0] : c.v>.44 ? PALETTE[5] : PALETTE[4];
      ctx.fillStyle=`rgba(${col[0]},${col[1]},${col[2]},${a})`;
      ctx.fillText(c.ch, ox+c.x*cell, oy+c.y*cell*.78);
    }
    ctx.restore();
  }

  function drawVoid(t) {
    const cx=W*.665, cy=H*.49;
    const minD=Math.min(W,H);
    const rx=minD*.145, ry=minD*.108;

    ctx.save();
    ctx.globalCompositeOperation='screen';
    let halo=ctx.createRadialGradient(cx,cy,rx*.55,cx,cy,rx*2.65);
    halo.addColorStop(0,'rgba(0,0,0,0)');
    halo.addColorStop(.28,'rgba(108,233,255,.035)');
    halo.addColorStop(.48,'rgba(255,71,206,.075)');
    halo.addColorStop(.72,'rgba(102,232,255,.032)');
    halo.addColorStop(1,'rgba(0,0,0,0)');
    ctx.fillStyle=halo; ctx.fillRect(cx-rx*3,cy-rx*3,rx*6,rx*6);
    ctx.restore();

    // Occluding body / machined absence.
    ctx.save();
    ctx.translate(cx,cy);
    ctx.scale(1,.76);
    const wobble = 1 + Math.sin(t*.13)*.008;
    let g=ctx.createRadialGradient(-rx*.12,-rx*.10,rx*.08,0,0,rx*1.18*wobble);
    g.addColorStop(0,'#06070c');
    g.addColorStop(.58,'#010207');
    g.addColorStop(.86,'#000106');
    g.addColorStop(1,'rgba(0,0,0,0)');
    ctx.fillStyle=g;
    ctx.beginPath(); ctx.ellipse(0,0,rx*1.08,rx*1.03,0,0,Math.PI*2); ctx.fill();

    ctx.strokeStyle='rgba(228,235,255,.15)';
    ctx.lineWidth=.6;
    ctx.setLineDash([1.5,7]);
    ctx.beginPath(); ctx.ellipse(0,0,rx*1.02,rx*.98,0,0,Math.PI*2); ctx.stroke();
    ctx.restore();
  }

  function drawScanCut(t) {
    // One slow moving inspection/cut line rather than rapid glitching.
    const y = ((t*.018)%1) * (H*1.18) - H*.09;
    ctx.save();
    ctx.globalCompositeOperation='screen';
    const g=ctx.createLinearGradient(0,y-16,0,y+16);
    g.addColorStop(0,'rgba(0,0,0,0)');
    g.addColorStop(.5,'rgba(211,242,255,.045)');
    g.addColorStop(1,'rgba(0,0,0,0)');
    ctx.fillStyle=g; ctx.fillRect(0,y-18,W,36);
    ctx.restore();
  }
