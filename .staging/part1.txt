tctx.setTransform(1,0,0,1,0,0);
    tctx.clearRect(0, 0, trails.width, trails.height);
    tctx.setTransform(DPR,0,0,DPR,0,0);

    const area = W * H;
    const count = clamp(Math.floor(area / 900), 700, 1900);
    particles = new Array(count);
    for (let i = 0; i < count; i++) particles[i] = spawnParticle(i, true);

    branches = makeBranches();
    toolpaths = makeToolpaths();
    glyphCells = makeGlyphCells();
    stateEl.textContent = `canonical seed 0x${seed.toString(16).toUpperCase().padStart(8,'0')} · fixed step`;

    // Arrive as an already-running world rather than an empty boot screen.
    // This is deterministic fixed-step history, not a random visual warm-up.
    for (let i = 0; i < 180; i++) simulate();
  }

  function spawnParticle(id, initial = false) {
    const edgeBias = rng();
    let x, y;
    if (edgeBias < .42) {
      x = rng() < .5 ? randRange(-30, W * .18) : randRange(W * .82, W + 30);
      y = randRange(-20, H + 20);
    } else if (edgeBias < .78) {
      x = randRange(-20, W + 20);
      y = rng() < .5 ? randRange(-20, H * .16) : randRange(H * .84, H + 20);
    } else {
      x = randRange(W * .05, W * .95);
      y = randRange(H * .05, H * .95);
    }
    return {
      id,
      x, y,
      px: x, py: y,
      vx: randRange(-.16, .16),
      vy: randRange(-.16, .16),
      age: initial ? Math.floor(rng() * 1200) : 0,
      life: Math.floor(randRange(600, 2600)),
      speed: randRange(.40, 1.28),
      width: randRange(.32, 1.25),
      chroma: Math.floor(rng() * PALETTE.length),
      phase: rng() * Math.PI * 2,
      kind: rng()
    };
  }

  function fieldAt(x, y, t, p) {
    const nx = (x / Math.max(W,1) - .5) * 2;
    const ny = (y / Math.max(H,1) - .5) * 2;

    // Slowly evolving interference field: plasma-like complexity without random flicker.
    const q =
      Math.sin(nx * 4.7 + t * .055) +
      Math.cos(ny * 6.2 - t * .041) +
      .55 * Math.sin((nx + ny) * 9.1 + t * .028) +
      .34 * Math.cos((nx * 1.8 - ny * 2.7) * 5.4 - t * .021);

    let fx = Math.cos(q * 1.17 + ny * 2.4);
    let fy = Math.sin(q * 1.23 - nx * 2.0);

    // Central excavated void: flow wraps around it instead of filling it.
    const cx = W * .665;
    const cy = H * .49;
    const minD = Math.min(W, H);
    const rx = minD * .145;
    const ry = minD * .108;
    const dx = (x - cx) / rx;
    const dy = (y - cy) / ry;
    const er = Math.sqrt(dx*dx + dy*dy) + .0001;
    const halo = Math.exp(-Math.pow((er - 1.16) / .52, 2));
    fx += (-dy / er) * halo * 2.3;
    fy += ( dx / er) * halo * 1.75;
    if (er < 1.05) {
      const push = (1.05 - er) * 5.2;
      fx += (dx / er) * push;
      fy += (dy / er) * push;
    }

    // A faint downward material bias on part of the population.
    if (p.kind < .18) fy += .58;

    // Pointer behaves like a temporary tool head, but fades back to canonical motion.
    if (pointer.strength > .001) {
      const pdx = x - pointer.x;
      const pdy = y - pointer.y;
      const pr2 = pdx*pdx + pdy*pdy + 180;
      const reach = Math.min(W,H) * .24;
      const a = Math.exp(-pr2 / (reach * reach)) * pointer.strength;
      fx += (-pdy / Math.sqrt(pr2)) * a * 3.0;
      fy += ( pdx / Math.sqrt(pr2)) * a * 3.0;
    }

    return [fx, fy];
  }

  function simulate() {
    const t = tick * STEP;
    pointer.strength *= .982;

    tctx.save();
    tctx.globalCompositeOperation = 'source-over';
    tctx.fillStyle = 'rgba(2,3,7,0.012)';
    tctx.fillRect(0, 0, W, H);
    tctx.globalCompositeOperation = 'lighter';

    for (let i = 0; i < particles.length; i++) {
      let p = particles[i];
      p.px = p.x; p.py = p.y;
      const f = fieldAt(p.x, p.y, t + p.phase * .015, p);

      const inertia = p.kind < .18 ? .90 : .935;
      p.vx = p.vx * inertia + f[0] * .074 * p.speed;
      p.vy = p.vy * inertia + f[1] * .074 * p.speed;

      const mag = Math.hypot(p.vx, p.vy) || 1;
      const maxV = 2.1 * p.speed + .9;
      if (mag > maxV) { p.vx *= maxV / mag; p.vy *= maxV / mag; }

      p.x += p.vx;
      p.y += p.vy;
      p.age++;

      const c = PALETTE[p.chroma];
      const speedAlpha = clamp(Math.hypot(p.vx,p.vy) / 2.6, .12, 1);
      const alpha = (.045 + .22 * speedAlpha) * (p.kind < .18 ? .75 : 1);

      if ((i & 7) === 0) {
        tctx.strokeStyle = `rgba(${c[0]},${c[1]},${c[2]},${alpha*.21})`;
        tctx.lineWidth = p.width * 5.5;
        tctx.beginPath(); tctx.moveTo(p.px,p.py); tctx.lineTo(p.x,p.y); tctx.stroke();
      }

      tctx.strokeStyle = `rgba(${c[0]},${c[1]},${c[2]},${alpha})`;
      tctx.lineWidth = p.width;
      tctx.beginPath(); tctx.moveTo(p.px,p.py); tctx.lineTo(p.x,p.y); tctx.stroke();

      // Sparse deposition cells — visible evidence of work rather than full-screen noise.
      if (((p.id + tick) & 31) === 0) {
        const g = Math.max(3, Math.min(W,H) / 170);
        const sx = Math.round(p.x / g) * g;
        const sy = Math.round(p.y / g) * g;
        tctx.fillStyle = `rgba(${c[0]},${c[1]},${c[2]},${alpha*.55})`;
